import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import make_classification
from sklearn.svm import SVC

# 生成二维数据（方便可视化）
X, y = make_classification(n_samples=200, n_features=2, n_redundant=0,
                           n_clusters_per_class=1, random_state=42)

# 定义绘图函数
def plot_svm_boundary(X, y, model, title):
    plt.figure(figsize=(8, 6))
    
    # 绘制决策边界
    x_min, x_max = X[:, 0].min() - 0.5, X[:, 0].max() + 0.5
    y_min, y_max = X[:, 1].min() - 0.5, X[:, 1].max() + 0.5
    xx, yy = np.meshgrid(np.arange(x_min, x_max, 0.02),
                         np.arange(y_min, y_max, 0.02))
    Z = model.predict(np.c_[xx.ravel(), yy.ravel()])
    Z = Z.reshape(xx.shape)
    plt.contourf(xx, yy, Z, alpha=0.3, cmap='RdBu')
    
    # 绘制样本点
    plt.scatter(X[:, 0], X[:, 1], c=y, cmap='RdBu', edgecolors='k', s=50)
    
    # 标记支持向量
    plt.scatter(model.support_vectors_[:, 0], model.support_vectors_[:, 1],
               s=100, facecolors='none', edgecolors='k', linewidths=1.5, label='支持向量')
    
    plt.title(title)
    plt.legend()
    plt.show()
    
    # 打印信息
    print(f"{title}: 支持向量数量 = {len(model.support_vectors_)}")
    print(f"训练准确率 = {model.score(X, y):.4f}")

# 实验不同C值
C_values = [0.1, 1, 10, 100]

for C in C_values:
    svm = SVC(kernel='linear', C=C, random_state=42)
    svm.fit(X, y)
    plot_svm_boundary(X, y, svm, f'C = {C}')
```

1. C 越小，间隔越大；C 越大，间隔越小
2. 支持向量数量与 C 负相关：C 越大，支持向量越少

# 生成非线性数据（环形数据）
from sklearn.datasets import make_circles

X, y = make_circles(n_samples=200, factor=0.5, noise=0.05, random_state=42)

# 实验不同gamma值
gamma_values = [0.1, 1, 10]

for gamma in gamma_values:
    svm = SVC(kernel='rbf', C=1.0, gamma=gamma, random_state=42)
    svm.fit(X, y)
    plot_svm_boundary(X, y, svm, f'gamma = {gamma}')
```

1. gamma 控制影响半径。gamma=0.1 时，每个样本影响范围大，边界平滑；gamma=10 时，每个样本只影响近邻，边界复杂
2. gamma 越大，决策边界越复杂

import numpy as np
import pandas as pd
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score
from sklearn.preprocessing import StandardScaler

# 模拟数据：增加refund_amount_90d
np.random.seed(42)
n_samples = 1000

# 原始特征
recency = np.random.exponential(30, n_samples)
frequency = np.random.poisson(5, n_samples)
monetary = np.random.gamma(2, 100, n_samples)
avg_order_value = monetary / (frequency + 0.01)

# 新增退款金额特征（与monetary相关但不完全一致）
refund_amount_90d = np.random.beta(1, 10, n_samples) * monetary * 0.3

# 创建DataFrame
df_new = pd.DataFrame({
    'recency': recency,
    'frequency': frequency,
    'monetary': monetary,
    'avg_order_value': avg_order_value,
    'refund_amount_90d': refund_amount_90d
})

# 标准化
features = ['recency', 'frequency', 'monetary', 'avg_order_value', 'refund_amount_90d']
scaler = StandardScaler()
X_new = scaler.fit_transform(df_new[features])

# 聚类（k=4）
model_new = KMeans(n_clusters=4, init="k-means++", n_init=20, random_state=42)
labels_new = model_new.fit_predict(X_new)

# 分析簇画像
centers_new = scaler.inverse_transform(model_new.cluster_centers_)
centers_df_new = pd.DataFrame(centers_new, columns=features)
print("增加退款金额后的簇画像：")
print(centers_df_new)

# 对比原模型（无退款特征）
df_old = df_new[['recency', 'frequency', 'monetary', 'avg_order_value']]
X_old = scaler.fit_transform(df_old)
model_old = KMeans(n_clusters=4, init="k-means++", n_init=20, random_state=42)
labels_old = model_old.fit_predict(X_old)

print(f"\n原模型轮廓系数: {silhouette_score(X_old, labels_old):.3f}")
print(f"新模型轮廓系数: {silhouette_score(X_new, labels_new):.3f}")

# 分析退款金额在各簇的分布
df_new['cluster'] = labels_new

def generate_medical_data(n_patients=1000, random_state=42):
    """生成模拟患者数据集"""
    np.random.seed(random_state)
    n = n_patients
    
    # 生成特征
    age = np.random.normal(55, 15, n).clip(25, 80)
    bmi = np.random.normal(26, 5, n).clip(18, 40)
    heart_rate = np.random.normal(75, 12, n).clip(55, 110)
    fasting_glucose = np.random.normal(5.8, 2.0, n).clip(4.0, 15.0)
    smoking = np.random.binomial(1, 0.25, n)
    exercise_hours = np.random.exponential(3, n).clip(0, 15)
    family_history = np.random.binomial(1, 0.35, n)
    
    # 生成目标变量（收缩压）
    sbp = (
        80.0 +                    # 基准血压
        0.55 * age +              # 年龄正相关
        1.30 * bmi +              # BMI正相关
        0.12 * heart_rate +       # 心率正相关
        1.80 * fasting_glucose +  # 血糖正相关
        5.00 * smoking +          # 吸烟升高血压
        -0.80 * exercise_hours +  # 运动降低血压
        4.50 * family_history +   # 家族史风险
        np.random.normal(0, 8, n) # 随机噪声
    ).clip(90, 210)
    
    # 整合为DataFrame
    df = pd.DataFrame({
        'age': age.round(1),
        'bmi': bmi.round(1),
        'heart_rate': heart_rate.round(0).astype(int),
        'fasting_glucose': fasting_glucose.round(1),
        'smoking': smoking,
        'exercise_hours': exercise_hours.round(1),
        'family_history': family_history,
        'sbp': sbp.round(1)
    })
    
    return df

def generate_medical_data_with_ckd(n_patients=1000, random_state=42):
    """生成包含CKD特征的模拟患者数据集"""
    np.random.seed(random_state)
    n = n_patients

    # 生成特征
    age = np.random.normal(55, 15, n).clip(25, 80)
    bmi = np.random.normal(26, 5, n).clip(18, 40)
    heart_rate = np.random.normal(75, 12, n).clip(55, 110)
    fasting_glucose = np.random.normal(5.8, 2.0, n).clip(4.0, 15.0)
    smoking = np.random.binomial(1, 0.25, n)
    exercise_hours = np.random.exponential(3, n).clip(0, 15)
    family_history = np.random.binomial(1, 0.35, n)

    # 新增：慢性肾病（CKD）特征
    # CKD与年龄、血压、血糖相关，发生率约10-15%
    ckd_prob = 0.05 + 0.002 * (age - 25) + 0.01 * (fasting_glucose - 5) + 0.005 * (bmi - 25)
    ckd_prob = np.clip(ckd_prob, 0.02, 0.35)
    ckd = np.random.binomial(1, ckd_prob, n)

    # 生成目标变量（收缩压）
    # CKD使血压额外升高约8-12 mmHg
    sbp = (
        80.0 +
        0.55 * age +
        1.30 * bmi +
        0.12 * heart_rate +
        1.80 * fasting_glucose +
        5.00 * smoking +
        -0.80 * exercise_hours +
        4.50 * family_history +
        9.50 * ckd +              # 新增：CKD升高血压约9.5 mmHg
        np.random.normal(0, 8, n)
    ).clip(90, 210)

    df = pd.DataFrame({
        'age': age.round(1),
        'bmi': bmi.round(1),
        'heart_rate': heart_rate.round(0).astype(int),
        'fasting_glucose': fasting_glucose.round(1),
        'smoking': smoking,
        'exercise_hours': exercise_hours.round(1),
        'family_history': family_history,
        'ckd': ckd,  # 新增特征
        'sbp': sbp.round(1)
    })

    return df
